      ---------------------------------------------------------------------------
      -- Custom non-Mason servers
      ---------------------------------------------------------------------------

      -- NimLSP installed through Nimble instead of Mason.
      --
      -- NimLSP accepts the Nim installation root as its first argument.
      -- Supplying it explicitly avoids NimLSP using the Nim installation
      -- path that was embedded when nimlsp.exe was compiled.
      do
        local nimlsp = vim.fn.exepath 'nimlsp'
        local nim = vim.fn.exepath 'nim'

        if nimlsp == '' then
          vim.schedule(function()
            vim.notify(
              'nimlsp not found in PATH. Run: nimble install nimlsp',
              vim.log.levels.WARN
            )
          end)
        elseif nim == '' then
          vim.schedule(function()
            vim.notify(
              'Nim compiler not found in PATH',
              vim.log.levels.WARN
            )
          end)
        else
          -- nim.exe is <nim_root>/bin/nim.exe
          local nim_bin = vim.fs.dirname(nim)
          local nim_root = vim.fs.dirname(nim_bin)

          vim.lsp.config('nimls', {
            cmd = {
              nimlsp,
              nim_root,
            },
            capabilities = vim.tbl_deep_extend('force', {}, capabilities),
          })

          vim.lsp.enable 'nimls'
        end
      end

      do
        local ok, ahk = pcall(require, 'custom.ahk2')
        if ok then
          local ahk_config, err = ahk.build_config()
          if ahk_config then
            ahk_config.capabilities = vim.tbl_deep_extend('force', {}, capabilities, ahk_config.capabilities or {})

            vim.lsp.config('ahk2', ahk_config)
            vim.lsp.enable 'ahk2'
          else
            vim.schedule(function()
              vim.notify(err, vim.log.levels.WARN)
            end)
          end
        end
      end
